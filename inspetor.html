async function enviarInspecao() { const nomeDigitado = document.getElementById('inspetorNome').value.trim(); const nomeFinal = nomeDigitado || 'Inspetor Anônimo'; const itens = checklistSel.itens; let conforme = 0, naoConforme = 0, temTexto = false; const respostasArr = itens.map(item => { const r = respostas[item.id] || ''; if (r === 'conforme') conforme++; else if (r === 'nao_conforme') naoConforme++; else if (item.tipo === 'texto' && r.trim() !== '') temTexto = true; return { item_id: item.id, pergunta: item.pergunta, tipo: item.tipo || 'conforme', resposta: r }; }); // ✅ A MÁGICA ACONTECE AQUI: Se tem texto, conta como 1 NC const totalNC = naoConforme + (temTexto ? 1 : 0); const totalItensAvaliativos = conforme + naoConforme; // Se não tiver itens de marcar, o percentual é 0 (pois teve uma falha) const percentualFinal = totalItensAvaliativos > 0 ? (conforme / totalItensAvaliativos) * 100 : (temTexto ? 0 : 100); const btn = document.getElementById('btnEnviar'); btn.disabled = true; btn.textContent = '⏳ Enviando...'; try { const { error } = await db.from('veiculos_inspecoes').insert([{ checklist_id: checklistSel.id, checklist_titulo: checklistSel.titulo, checklist_categoria: checklistSel.categoria, inspetor_nome: nomeFinal, veiculo_placa: veiculoSel.placa, veiculo_modelo: 'N/A', local_inspecionado: document.getElementById('localInsp').value.trim(), respostas: respostasArr, fotos: fotos, total_itens: itens.length, total_conformes: conforme, total_nao_conformes: totalNC, percentual_conformidade: percentualFinal, status: 'concluida', status_resolucao: 'pendente' }]); if (error) throw error; ultimaInspecao = { veiculo: veiculoSel.placa, checklist: checklistSel.titulo, inspetor: nomeFinal, local: document.getElementById('localInsp').value.trim(), data: new Date().toLocaleString('pt-BR'), itens: itens, respostas: respostasArr, fotos: fotos, conforme, naoConforme: totalNC, percentual: percentualFinal }; document.getElementById('resumoFinal').innerHTML = `
${conforme}
CONFORMES
${totalNC}
NÃO CONFORMES
`; irParaEtapa(4); mostrarToast('✅ Inspeção enviada com sucesso!'); } catch (err) { console.error(err); mostrarToast('❌ ' + err.message, true); btn.disabled = false; atualizarStatusBotao(); } }
